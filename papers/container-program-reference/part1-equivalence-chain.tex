\begin{abstract}
This is Part~I of the comprehensive, registry-grounded technical record of the Kodamai
container program. It develops the theory core: the category of containers and its
representation theorem, the four canonical monoidal structures on $\mathrm{Cont}/\mathrm{Poly}$,
the closed structures and Day convolution, change of base over a general category, the free
monad and cofree comonad, and the equivalence chain $\mathrm{Cont}\supseteq\mathrm{DCont}\simeq
\text{Poly-comonoid}\simeq\mathrm{Cat}$ with honest status on which links are theorems and which
are conjectural. Every result carries its \textbf{trust grade in brackets}, copied verbatim from
the proof registry, together with its registry node identifier. Grades are never inflated.
\end{abstract}

\paragraph{Thesis of Part I.}
A container is a family of sets, $\mathrm{Cont}\cong\mathrm{Fam}(\mathrm{Set}^{op})$; the four
canonical tensors $(+,\times,\otimes,\triangleleft)$ all have the form ``shapes multiply, the
base's structure appears only in positions''; and the sequential tensor $\triangleleft$ is the one
whose comonoids are small categories. This is the convergence hub
$\mathrm{Cont}\supseteq\mathrm{DCont}\simeq\text{Poly-comonoid}\simeq\mathrm{Cat}$, developed with
honest status on which links are theorems and which are conjectural.

which everything else rests.

\section{Closed structures and Day convolution}
\emph{Precis.} Which tensors are closed, the internal-hom formulas, and the obstructions.
$(\mathrm{Poly},\times)$ is cartesian closed; $\mathrm{Cont}$ is CCC but not locally cartesian
closed. Day convolution and its coend-free degeneration on $\mathrm{Cont}$.

\begin{theorem}[Closed Day structures, \grade{proved}]
A Day/convolutional tensor $(\mathrm{Cont},⊙_⋆)$ is left-closed iff $(-)⋆B$ is polynomial for
every $B$, with a uniform $Π$-formula internal hom. Node \node{closed-day-structures}.
\end{theorem}

\begin{theorem}[Closed-tensor classification, \grade{in-progress}]
Classification of left-closed convolutional tensors on $\mathrm{Set}$: bounded arity implies
$\times$ or $∨_S$. The \textbf{infinite-arity boundary is OPEN}. Node
\node{closed-tensor-classification} --- one of the program's two standing open fronts.
\end{theorem}

\begin{proposition}[Dirichlet internal hom, \grade{lean-verified}]
$(\mathrm{Cont},\otimes,y)$ is closed with the Dirichlet internal hom (\texttt{DirichletClosed.lean},
\texttt{DirichletHomPi.lean}). The mathematics is Niu--Spivak prior art; the proof object is ours.
\end{proposition}
